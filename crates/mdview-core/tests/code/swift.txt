import Foundation

@MainActor
protocol Shape { var area: Double { get } }

struct Square: Shape {
    let side: Double
    var area: Double { side * side }
}

func describe(_ s: some Shape) -> String {
    let multi = """
    Area: \(s.area)
    """
    guard s.area > 0 else { return "empty" }
    return multi // done
}

let shapes: [Shape] = [Square(side: 2)]
print(shapes.map { describe($0) }, nil, self)
