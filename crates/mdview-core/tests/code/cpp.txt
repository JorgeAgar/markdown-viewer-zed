#pragma once
#include <vector>

namespace app {
template <typename T>
class Widget final : public Base {
public:
    explicit Widget(std::vector<T> items) noexcept : items_(std::move(items)) {}
    virtual ~Widget() override = default;
    constexpr static int kSize = 0b1010;
    auto get() const -> const T& { return items_[0]; }
private:
    std::vector<T> items_;
};
}  // namespace app

int main() {
    auto w = new Widget<int>({1, 2, 3});
    if (w != nullptr) delete w;
    co_return;
}
